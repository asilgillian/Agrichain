@baseUrl = https://localhost:7001
@token = YOUR_JWT_TOKEN_HERE

### Login — run this first, copy token into @token above
POST {{baseUrl}}/api/auth/login
Content-Type: application/json

{
  "email": "admin@pml.com",
  "password": "Admin@1234"
}

###

### Get all deliveries
GET {{baseUrl}}/api/logistics/deliveries
Authorization: Bearer {{token}}
Accept: application/json

###

### Create a delivery
POST {{baseUrl}}/api/logistics/deliveries
Authorization: Bearer {{token}}
Content-Type: application/json

{
  "customerCode": "CUST001",
  "customerName": "Kampala Pharmacy",
  "deliveryDate": "2025-04-20",
  "assignedDriverID": null
}

###

### Get all incidents
GET {{baseUrl}}/api/incidents
Authorization: Bearer {{token}}

###

### Report an incident
POST {{baseUrl}}/api/incidents
Authorization: Bearer {{token}}
Content-Type: application/json

{
  "title": "Spill in warehouse aisle 3",
  "description": "Chemical spill near storage bay",
  "severity": "Minor",
  "location": "Warehouse - Aisle 3"
}
